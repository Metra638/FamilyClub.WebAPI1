helm upgrade --install ingress-nginx-prod ingress-nginx/ingress-nginx \
  --namespace ingress-nginx-prod --create-namespace \
  --set controller.serviceAccount.name=ingress-nginx \
  --set controller.ingressClassResource.name=nginx-prod \
  --set controller.ingressClassResource.controllerValue=k8s.io/ingress-nginx-prod

  helm upgrade --install ingress-nginx-stag ingress-nginx/ingress-nginx \
  --namespace ingress-nginx-stag --create-namespace \

  --set controller.ingressClassResource.name=nginx-stag \
